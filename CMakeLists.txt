cmake_minimum_required(VERSION 3.16)
project(cpp_programming_fundamentals CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Boost QUIET)
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/0*/*.cpp")

foreach(src ${SOURCES})
  get_filename_component(name ${src} NAME_WE)
  file(READ ${src} contents)
  if(contents MATCHES "boost/" AND NOT Boost_FOUND)
    message(STATUS "Skipping ${name} (requires Boost)")
    continue()
  endif()
  add_executable(${name} ${src})
  if(contents MATCHES "boost/")
    target_include_directories(${name} PRIVATE ${Boost_INCLUDE_DIRS})
  endif()
endforeach()
